    let currentQuestion = 0;
    let selectedQuestions = [];
    let scores = { a: 0, b: 0, o: 0 };
    const TOTAL_QUESTIONS = 12;

    // 洗牌算法抽取12题
    function shuffleAndPick() {
        let shuffled = [...questionBank].sort(() => 0.5 - Math.random());
        return shuffled.slice(0, TOTAL_QUESTIONS);
    }

    function startQuiz() {
        currentQuestion = 0;
        scores = { a: 0, b: 0, o: 0 };
        selectedQuestions = shuffleAndPick();
        showSection('quiz');
        renderQuestion();
    }

    function renderQuestion() {
        const qData = selectedQuestions[currentQuestion];
        
        document.getElementById('progress-text').innerText = `${currentQuestion + 1} / ${TOTAL_QUESTIONS}`;
        document.getElementById('progress-bar').style.width = `${((currentQuestion + 1) / TOTAL_QUESTIONS) * 100}%`;

        let html = `<h3 class="question-title">${qData.q}</h3><div class="options-list">`;
        qData.opts.forEach((opt, index) => {
            html += `<button class="option-btn" onclick="selectOption(${index})">${opt.text}</button>`;
        });
        html += `</div>`;
        
        const area = document.getElementById('question-area');
        area.style.opacity = 0;
        setTimeout(() => {
            area.innerHTML = html;
            area.style.opacity = 1;
            area.style.transition = 'opacity 0.3s';
        }, 200);
    }

    function selectOption(index) {
        const optScores = selectedQuestions[currentQuestion].opts[index].scores;
        scores.a += optScores.a;
        scores.b += optScores.b;
        scores.o += optScores.o;

        currentQuestion++;
        if (currentQuestion < TOTAL_QUESTIONS) {
            renderQuestion();
        } else {
            calculateResult();
        }
    }

    function calculateResult() {
        showSection('loading');
        
        setTimeout(() => {
            const { a, b, o } = scores;
            const maxScore = Math.max(a, b, o);
            
            let resultKey = "";

            if (a > 25) {
                resultKey = "Enigma"; 
            } else if (Math.abs(a - b) <= 3 && Math.abs(b - o) <= 3 && Math.abs(a - o) <= 3) {
                resultKey = "Delta"; 
            } else if (maxScore === a) {
                if (b > o) resultKey = "Sigma"; 
                else resultKey = "Alpha";
            } else if (maxScore === b) {
                resultKey = "Beta";
            } else if (maxScore === o) {
                resultKey = "Omega";
            }

            renderResult(resultKey);
        }, 2500);
    }

    function renderResult(key) {
        const res = results[key];
        let statsHtml = "";
        for (const [label, value] of Object.entries(res.stats)) {
            statsHtml += `
                <div class="stat-item">
                    <div class="stat-label">${label}</div>
                    <div class="stat-value">${value}</div>
                </div>`;
        }

        const html = `
            <span class="emoji-large">${res.emoji}</span>
            <div class="result-badge">${res.badge}</div>
            <h2 class="result-title">${res.title}</h2>
            <p class="result-pheromone">${res.pheromone}</p>
            <div class="result-desc">${res.desc}</div>
            <div class="stats-grid">${statsHtml}</div>
            <button class="btn" onclick="startQuiz()">重新测试 🔄</button>
            <button class="btn btn-outline" onclick="showSection('password-gate')">返回首页</button>
        `;

        document.getElementById('result-content').innerHTML = html;
        showSection('result');
    }
